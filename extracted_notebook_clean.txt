        # Get all images in the training subfolder        images = os.listdir(train_subfolder_path)        total_images = len(images)
        # Calculate number of images to move        num_images_to_move = int(total_images * val_percentage)
        # Randomly select images to move        images_to_move = random.sample(images, num_images_to_move)
        # Move images        for image in images_to_move:            src_path = os.path.join(train_subfolder_path, image)            dest_path = os.path.join(val_subfolder_path, image)            shutil.move(src_path, dest_path)

def attention_block(x, g, inter_channel):    theta_x = Conv2D(inter_channel, (1, 1), padding='same')(x)    phi_g = Conv2D(inter_channel, (1, 1), padding='same')(g)    f = tf.keras.layers.Activation('relu')(tf.keras.layers.Add()([theta_x, phi_g]))    psi = Conv2D(1, (1, 1), padding='same', activation='sigmoid')(f)    return tf.keras.layers.Multiply()([x, psi])
