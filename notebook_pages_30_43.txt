def residual_block(x, filters):    shortcut = Conv2D(filters, (1,1), padding='same')(x)  # Ensure the shortcut has the same channels    x = Conv2D(filters, (3,3), padding='same')(x)    x = BatchNormalization()(x)    x = Activation('relu')(x)    x = Conv2D(filters, (3,3), padding='same')(x)    x = BatchNormalization()(x)    x = Add()([x, shortcut])  # Residual Connection    x = Activation('relu')(x)    return x
